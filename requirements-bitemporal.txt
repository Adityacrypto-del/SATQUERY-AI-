# Branch-2 (bi-temporal change analysis) dependencies.

# Named for the branch rather than called requirements.txt so it cannot
# collide with the single-image branch's own file on merge.
#
# Versions are the ones this branch was developed and measured against, not

# numbers.

# -- analysis core ---------------------------------------------------------

scipy==1.17.1

# https://pytorch.org rather than taking the default wheel blindly.
torch==2.6.0
torchvision==0.21.0

# parquet. Not needed to run the specialist.

# -- tests -----------------------------------------------------------------
